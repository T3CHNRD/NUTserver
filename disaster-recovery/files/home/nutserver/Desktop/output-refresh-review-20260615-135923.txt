     2	import json

     9	LOG = Path("/var/log/nut-orchestrator-ui/power-events.log")
    10	NOW = datetime.now().astimezone()
    11	CUTOFF = NOW - timedelta(days=5)
    12	
    13	LOG_TS_RE = re.compile(r'^\[(?P<ts>[0-9]{4}-[0-9]{2}-[0-9]{2}(?:[ T])[0-9]{2}:[0-9]{2}:[0-9]{2}(?:[+-][0-9]{2}:?[0-9]{2})?)\]')
    14	JOURNAL_TS_RE = re.compile(r'^(?P<mon>[A-Z][a-z]{2})\s+(?P<day>[0-9]{1,2})\s+(?P<time>[0-9]{2}:[0-9]{2}:[0-9]{2})\s+')
    15	
    16	INTERESTING_JOURNAL_RE = re.compile(
    17	    r'(on battery|on line power|battery is low|low battery|upssched|exec_cmd|permission denied|returned|ups[0-9])',
    18	    re.IGNORECASE
    19	)
    20	
    21	MONTHS = {
    22	    "Jan": 1, "Feb": 2, "Mar": 3, "Apr": 4,
    23	    "May": 5, "Jun": 6, "Jul": 7, "Aug": 8,
    24	    "Sep": 9, "Oct": 10, "Nov": 11, "Dec": 12,
    25	}
    26	
    27	def parse_log_ts(line):
    28	    m = LOG_TS_RE.match(line)
    29	    if not m:
    30	        return None
    31	
    32	    raw = m.group("ts").replace("T", " ")
    33	
    34	    if re.search(r'[+-][0-9]{2}:[0-9]{2}$', raw):
    35	        raw = raw[:-3] + raw[-2:]
    36	
    37	    for fmt in ("%Y-%m-%d %H:%M:%S%z", "%Y-%m-%d %H:%M:%S"):
    38	        try:
    39	            dt = datetime.strptime(raw, fmt)
    40	            if dt.tzinfo is None:
    41	                dt = dt.replace(tzinfo=NOW.tzinfo)
    42	            return dt.astimezone(NOW.tzinfo)
    43	        except ValueError:
    44	            pass
    45	
    46	    return None
    47	
    48	def parse_journal_ts(line):
    49	    m = JOURNAL_TS_RE.match(line)
    50	    if not m:
    51	        return None
    52	
    53	    try:
    54	        month = MONTHS[m.group("mon")]
    55	        day = int(m.group("day"))
    56	        hour, minute, second = map(int, m.group("time").split(":"))
    57	        dt = datetime(NOW.year, month, day, hour, minute, second, tzinfo=NOW.tzinfo)
    58	
    59	        # Handle year boundary safely if journal line is from Dec while now is Jan.
    60	        if dt > NOW + timedelta(days=1):
    61	            dt = dt.replace(year=NOW.year - 1)
    62	
    63	        return dt
    64	    except Exception:
    65	        return None
    66	
    67	def normalize_journal_line(line, dt):
    68	    # Keep the original journal message, but put it into the same bracketed timestamp format
    69	    # so both Dashboard and NUT UI parse/render it consistently.
    70	    msg = JOURNAL_TS_RE.sub("", line).strip()
    71	    return f"[{dt.strftime('%Y-%m-%d %H:%M:%S')}] NUT_MONITOR_EVENT {msg}"
    72	
    73	events = []
    74	
    75	# Source 1: orchestrator/UI power log
    76	try:

    93	        text=True,

    95	        stderr=subprocess.DEVNULL,
    96	        check=False,
    97	    )
    98	
    99	    for idx, line in enumerate(result.stdout.splitlines()):
   100	        if not INTERESTING_JOURNAL_RE.search(line):
   101	            continue
   102	
   103	        dt = parse_journal_ts(line)
   104	        if dt and dt >= CUTOFF:
   105	            events.append({
   106	                "dt": dt,
   107	                "idx": idx,
   108	                "source": "journal",

   110	            })
   111	except Exception:
   112	    pass
